cmake_minimum_required(VERSION 3.25)
project(gseek-inference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT LLAMA_SOURCE)
  message(FATAL_ERROR "Use scripts/build-inference.sh to obtain the pinned llama.cpp source")
endif()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
foreach(option COMMON TOOLS SERVER EXAMPLES TESTS APP UI)
  set(LLAMA_BUILD_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(gseek-inference main.cpp)
target_include_directories(gseek-inference PRIVATE "${LLAMA_SOURCE}/vendor" "${LLAMA_SOURCE}/tools/mtmd")
target_link_libraries(gseek-inference PRIVATE llama mtmd ggml ggml-base)
set_target_properties(gseek-inference PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
install(TARGETS gseek-inference RUNTIME DESTINATION .)
